  function rotateShape(shape) {
    var newShape = [];
    for (var i = 0; i < 4; i++) {
      newShape[i] = [];
      for (var j = 0; j < 4; j++) {
        newShape[i][j] = shape[3 - j][i];
      }
    }
    return newShape;
  }

  function collide(b, s, dx, dy, id) {
    for (var x = 0; x < 4; x++) {
      for (var y = 0; y < 4; y++) {
        if (s[x][y]) {
          var bx = x + dx, by = y + dy;
          if (bx < 0 || bx >= COLS || by < 0 || by >= ROWS) return true;
          if (b[by][bx] !== 0 && b[by][bx] !== id) return true;
        }
      }
    }
    return false;
  }

  function drawBoard() {
    ctx.fillStyle = '#111';
    ctx.fillRect(0, 0, COLS * 20, ROWS * 20);
    ctx.fillStyle = '#333';
    for (var y = 0; y < ROWS; y++) {
      for (var x = 0; x < COLS; x++) {
        if (board[y][x] !== 0) {
          ctx.fillStyle = COLORS[String(board[y][x])];
          ctx.fillRect(x * 20, y * 20, 20, 20);
        }
      }
    }
  }

  function drawTetromino() {
    if (!shape) return;
    var s = SHAPES[String(shapeId)][rotation];
    ctx.fillStyle = COLORS[String(shapeId)];
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 2;
    for (var y = 0; y < 4; y++) {
      for (var x = 0; x < 4; x++) {
        if (s[x][y]) {
          ctx.fillRect((x + shape.x) * 20, (y + shape.y) * 20, 20, 20);
          ctx.strokeRect((x + shape.x) * 20, (y + shape.y) * 20, 20, 20);
        }
      }
    }
  }

  function clearLines() {
    var linesCleared = 0;
    for (var y = 0; y < ROWS; y++) {
      if (board[y].every(function (c) { return c !== 0; })) {
        linesCleared++;
        board.splice(y, 1, Array(COLS).fill(0));
        board.unshift(Array(COLS).fill(0));
        y--;
      }
    }
    if (linesCleared) {
      score += linesCleared * 100 * level;
      lines += linesCleared;
      if (lines >= 10) { level++; lines = 0; dropInterval = Math.max(120, 800 - (level - 1) * 60); }
      statusEl.textContent = 'Zeilen: ' + linesCleared + ' | Punkte: ' + score + ' | Level: ' + level;
    }
  }

  function hardDrop() {
    while (!collide(board, SHAPES[String(shapeId)][rotation], shape.x, shape.y + 1, shapeId)) {
      shape.y++;
      dropCounter = 0;
    }
    shape.y--;
    if (!collide(board, SHAPES[String(shapeId)][rotation], shape.x, shape.y + 1, shapeId)) {
      shape.y++;
    }
    dropCounter = 0;
  }

  function move(dx, dy) {
    if (!shape) return;
    if (!collide(board, SHAPES[String(shapeId)][rotation], shape.x + dx, shape.y + dy, shapeId)) {
      shape.x += dx; shape.y += dy;
    }
    if (!collide(board, SHAPES[String(shapeId)][rotation], shape.x, shape.y + 1, shapeId)) {
      shape.y++;
    }
  }

  function rotate() {
    if (!shape) return;
    var s = rotateShape(SHAPES[String(shapeId)][rotation]);
    var dx = 0, dy = 0;
    if (collide(board, s, shape.x, shape.y, shapeId)) {
      if (shape.x < 3 && !collide(board, s, shape.x - 1, shape.y, shapeId)) dx = -1;
      if (shape.x > COLS - 4 && !collide(board, s, shape.x + 1, shape.y, shapeId)) dx = 1;
      if (shape.y < 3 && !collide(board, s, shape.x, shape.y - 1, shapeId)) dy = -1;
      if (shape.y > ROWS - 4 && !collide(board, s, shape.x, shape.y + 1, shapeId)) dy = 1;
    }
    if (!collide(board, s, shape.x + dx, shape.y + dy, shapeId)) {
      rotation = (rotation + 1) % 4;
      shape.x += dx; shape.y += dy;
    }
  }

  function spawn() {
    board = [];
    for (var y = 0; y < ROWS; y++) board.push(Array(COLS).fill(0));
    shape = null;
    shapeId = nextShapeId;
    nextShapeId = Math.floor(Math.random() * 7) + 1;
    rotation = 0;
    shape.x = COLS / 2;
    shape.y = 0;
    dropCounter = 0;
    dropInterval = Math.max(120, 800 - (level - 1) * 60);
    if (gameOver) {
      gameOver = false;
      overlay.classList.add('hidden');
      scoreEl.textContent = 'Punkte: ' + score;
      statusEl.textContent
